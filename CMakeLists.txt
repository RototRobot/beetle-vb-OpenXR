cmake_minimum_required(VERSION 3.24)
project(beetle_vb_openxr VERSION 0.1.0 LANGUAGES C CXX)

option(BVB_BUILD_OPENXR "Build OpenXR diagnostics and the Windows stereo screen test" ON)
include(CTest)

# Match the upstream Makefile.common source list and definitions.
add_library(beetle_vb_core SHARED
  libretro.cpp mednafen/mempatcher.cpp mednafen/hw_cpu/v810/v810_cpu.cpp
  mednafen/vb/vsu.c mednafen/vb/input.c mednafen/vb/timer.c mednafen/vb/vip.c
  mednafen/hw_cpu/v810/fpu-new/softfloat.c mednafen/sound/Blip_Buffer.c
  mednafen/state.c mednafen/settings.c
  libretro-common/compat/compat_strl.c libretro-common/compat/compat_snprintf.c)
target_include_directories(beetle_vb_core PRIVATE
  . mednafen mednafen/include mednafen/hw_sound mednafen/hw_cpu
  mednafen/hw_misc libretro-common/include)
target_compile_definitions(beetle_vb_core PRIVATE
  WANT_32BPP FRONTEND_SUPPORTS_RGB565 STDC_HEADERS __STDC_LIMIT_MACROS
  __LIBRETRO__ MEDNAFEN_VERSION="0.9.31" MEDNAFEN_VERSION_NUMERIC=931)
if(MSVC)
  target_compile_definitions(beetle_vb_core PRIVATE INLINE=_inline
    _CRT_SECURE_NO_DEPRECATE NOMINMAX)
else()
  target_compile_definitions(beetle_vb_core PRIVATE INLINE=inline)
  target_link_libraries(beetle_vb_core PRIVATE m)
endif()
set_target_properties(beetle_vb_core PROPERTIES PREFIX "" OUTPUT_NAME "mednafen_vb_libretro")

add_library(bvb_stereo STATIC frontend/stereo_frame.cpp)
target_include_directories(bvb_stereo PUBLIC frontend)
target_compile_features(bvb_stereo PUBLIC cxx_std_17)

add_library(bvb_presentation STATIC frontend/presentation.cpp)
target_link_libraries(bvb_presentation PUBLIC bvb_stereo)

add_library(bvb_core_host STATIC frontend/core_host.cpp frontend/game_input.cpp frontend/controller_bindings.cpp frontend/save_ram.cpp frontend/state_file.cpp)
target_include_directories(bvb_core_host PRIVATE libretro-common/include)
target_link_libraries(bvb_core_host PUBLIC bvb_stereo PRIVATE ${CMAKE_DL_LIBS})

add_library(bvb_menu STATIC frontend/settings_menu.cpp frontend/ui_draw.cpp)
target_link_libraries(bvb_menu PUBLIC bvb_presentation PRIVATE bvb_core_host)

add_library(bvb_vr_input STATIC frontend/vr_input.cpp)
target_link_libraries(bvb_vr_input PUBLIC bvb_core_host)

add_executable(bvb_core_probe frontend/core_probe.cpp)
target_include_directories(bvb_core_probe PRIVATE libretro-common/include)
target_link_libraries(bvb_core_probe PRIVATE bvb_core_host)
add_dependencies(bvb_core_probe beetle_vb_core)

if(BVB_BUILD_OPENXR)
  include(FetchContent)
  set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(BUILD_API_LAYERS OFF CACHE BOOL "" FORCE)
  set(BUILD_CONFORMANCE_TESTS OFF CACHE BOOL "" FORCE)
  set(BUILD_LOADER ON CACHE BOOL "" FORCE)
  set(DYNAMIC_LOADER OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(openxr_sdk
    GIT_REPOSITORY https://github.com/KhronosGroup/OpenXR-SDK.git
    GIT_TAG release-1.1.54
    GIT_SHALLOW TRUE
    GIT_CONFIG http.sslBackend=openssl)
  FetchContent_MakeAvailable(openxr_sdk)
  add_executable(bvb_xr_probe frontend/openxr_probe.cpp)
  target_compile_features(bvb_xr_probe PRIVATE cxx_std_17)
  target_link_libraries(bvb_xr_probe PRIVATE OpenXR::openxr_loader)
  if(WIN32)
    target_compile_definitions(bvb_xr_probe PRIVATE XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11 NOMINMAX)
    target_link_libraries(bvb_xr_probe PRIVATE d3d11 dxgi)
    add_library(bvb_openxr STATIC frontend/openxr_session.cpp frontend/openxr_input.cpp)
    target_compile_definitions(bvb_openxr PRIVATE XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11 NOMINMAX)
    target_link_libraries(bvb_openxr PUBLIC bvb_presentation bvb_vr_input PRIVATE OpenXR::openxr_loader d3d11 dxgi)
    add_executable(bvb_stereo_test frontend/stereo_test.cpp)
    target_compile_definitions(bvb_stereo_test PRIVATE NOMINMAX UNICODE _UNICODE)
    add_library(bvb_windows STATIC frontend/desktop_window.cpp frontend/audio_output.cpp)
    target_compile_definitions(bvb_windows PRIVATE NOMINMAX UNICODE _UNICODE)
    target_link_libraries(bvb_windows PUBLIC bvb_presentation PRIVATE user32 gdi32 xaudio2 ole32)
    add_library(bvb_miniz STATIC third_party/miniz/miniz.c third_party/miniz/miniz_tinfl.c third_party/miniz/miniz_zip.c)
    target_include_directories(bvb_miniz PUBLIC third_party/miniz)
    target_compile_definitions(bvb_miniz PUBLIC MINIZ_NO_DEFLATE_APIS MINIZ_NO_ZLIB_APIS MINIZ_NO_STDIO)
    add_library(bvb_library STATIC frontend/library.cpp frontend/cover_image.cpp frontend/game_data.cpp)
    target_compile_definitions(bvb_library PRIVATE NOMINMAX UNICODE _UNICODE)
    target_link_libraries(bvb_library PUBLIC bvb_menu PRIVATE bvb_miniz windowscodecs ole32 winhttp xmllite shlwapi)
    target_compile_options(bvb_library PRIVATE /W4)
    target_link_libraries(bvb_stereo_test PRIVATE bvb_openxr bvb_windows)
    add_executable(beetle_vb_openxr frontend/player.cpp)
    target_compile_definitions(beetle_vb_openxr PRIVATE NOMINMAX UNICODE _UNICODE)
    target_link_libraries(beetle_vb_openxr PRIVATE bvb_openxr bvb_windows bvb_core_host bvb_library xinput9_1_0)
    add_dependencies(beetle_vb_openxr beetle_vb_core)
    target_compile_options(bvb_openxr PRIVATE /W4)
    target_compile_options(bvb_stereo_test PRIVATE /W4)
    target_compile_options(bvb_windows PRIVATE /W4)
    target_compile_options(beetle_vb_openxr PRIVATE /W4)
  endif()
endif()
